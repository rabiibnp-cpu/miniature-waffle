#!/usr/bin/env bash
# ingestion_to_db.sh
# clush (tous les serveurs) -> parsing des logs ingestion -> insertion PostgreSQL (aps.ingestion_metrics)
# Une ligne deja presente (meme ts + job_name + pid) n'est JAMAIS reinseree
# (INSERT ... ON CONFLICT DO NOTHING sur la cle primaire).
#
# Cron (toutes les 5 min) :
#   */5 * * * * /chemin/ingestion_to_db.sh >> /chemin/ingestion_to_db.log 2>&1
#
# Prerequis : clush, gawk, psql
# Mot de passe : NE PAS le mettre dans le script -> fichier ~/.pgpass (chmod 600), ligne :
#   HOTE:5432:newh_preprod:newh_preprod_admin:LE_MOT_DE_PASSE

export PATH="/usr/local/bin:/usr/bin:/bin:$PATH"

# ---------- Configuration ----------
export PGHOST="${PGHOST:?Definir PGHOST (hote PostgreSQL)}"
export PGPORT="${PGPORT:-5432}"
export PGDATABASE="${PGDATABASE:-newh_preprod}"
export PGUSER="${PGUSER:-newh_preprod_admin}"
TABLE="${TABLE:-aps.ingestion_metrics}"
DB_TZ="${DB_TZ:-Europe/Paris}"          # fuseau des dates ecrites dans les logs
LOGDIR="${LOGDIR:-/logs/ingestion}"
CLUSH_OPTS="${CLUSH_OPTS:--a}"
LOOKBACK_DAYS="${LOOKBACK_DAYS:-1}"     # aujourd'hui + N jours precedents (gere le passage de minuit)

# ---------- Anti-chevauchement (si un run dure plus de 5 min) ----------
exec 9>/tmp/ingestion_to_db.lock
flock -n 9 || { echo "$(date '+%F %T') deja en cours, on saute"; exit 0; }

# ---------- Fenetre de dates ----------
DAYS="$(date +%F)"
for i in $(seq 1 "$LOOKBACK_DAYS"); do DAYS="$DAYS|$(date -d "-$i day" +%F)"; done
PATTERN="^($DAYS).*DriverTaskQueueService"
MTIME=$((LOOKBACK_DAYS + 1))            # ne lire que les fichiers modifies recemment

AWKFILE="$(mktemp)"
CSV="$(mktemp --suffix=.csv)"
trap 'rm -f "$AWKFILE" "$CSV"' EXIT

cat > "$AWKFILE" <<'AWK_EOF'
function secs(s,   t, n, u) {
  t = 0;
  while (match(s, /^[0-9]+[DHMS]/)) {
    n = substr(s, 1, RLENGTH-1) + 0;
    u = substr(s, RLENGTH, 1);
    t += n * (u=="D" ? 86400 : (u=="H" ? 3600 : (u=="M" ? 60 : 1)));
    s = substr(s, RLENGTH+1);
  }
  return t;
}
BEGIN {
  print "date,job,pid,workers_running,workers_failed,items_submitted,items_inflight,results_received,sub_results_received,errors,time_since_submitted,time_since_first_worker,time_since_last_result,throughput";
}
{
  gsub(/\r/, "");
  gsub(/\([+-][0-9]+\)/, "");
  if (!match($0, /([0-9]{4}-[0-9]{2}-[0-9]{2} [0-9:]{8}),[0-9]+ .*DriverTaskQueueService - \[([^#]+)#([^]]+)\] w:([0-9]+)\/([0-9]+) - i:([0-9]+)\/([0-9]+) - r:([0-9]+)\/([0-9]+)\/([0-9]+) - t:([^\/]+)\/([^\/]+)\/([^ ]+)/, m)) next;
  d = m[1]; pid = m[3];
  x = d; gsub(/[-:]/, " ", x);
  ts = mktime(x);
  thr = 0;
  if ((pid in pfl) && ts > pts[pid]) thr = (pfl[pid] - m[7]) / (ts - pts[pid]);
  pfl[pid] = m[7];
  pts[pid] = ts;
  printf "%s,%s,%s,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%.2f\n", d, m[2], pid, m[4], m[5], m[6], m[7], m[8], m[9], m[10], secs(m[11]), secs(m[12]), secs(m[13]), thr;
}
AWK_EOF

# ---------- 1) Collecte + parsing -> CSV temporaire ----------
REMOTE="find $LOGDIR -name 'ingestion.log*' -mtime -$MTIME -print0 | xargs -0 -r zgrep -h -E '$PATTERN' 2>/dev/null"

clush $CLUSH_OPTS -N "$REMOTE" \
  | LC_ALL=C sort -s -k1,2 \
  | gawk -f "$AWKFILE" > "$CSV"

NBCSV=$(( $(wc -l < "$CSV") - 1 ))
if [ "$NBCSV" -le 0 ]; then
  echo "$(date '+%F %T') aucune ligne trouvee (fenetre: $DAYS)"
  exit 0
fi

# ---------- 2) Insertion PostgreSQL (staging + ON CONFLICT DO NOTHING) ----------
INSERTED=$(set -o pipefail; psql -X -q -t -A -v ON_ERROR_STOP=1 <<SQL | tail -n 1
BEGIN;
CREATE TEMP TABLE stg (
  ts timestamp, job_name text, pid text,
  workers_running int, workers_failed int,
  items_submitted int, items_inflight int,
  results_received int, sub_results_received int, results_errors int,
  time_since_sub int, time_since_first int, time_since_last int,
  throughput numeric
) ON COMMIT DROP;
\copy stg FROM '$CSV' WITH (FORMAT csv, HEADER true)
WITH ins AS (
  INSERT INTO $TABLE
    (ts, job_name, pid, workers_running, workers_failed, items_submitted, items_inflight,
     results_received, results_errors, time_since_sub, time_since_first, time_since_last, throughput)
  SELECT ts AT TIME ZONE '$DB_TZ', job_name, pid, workers_running, workers_failed, items_submitted, items_inflight,
         results_received, results_errors, time_since_sub, time_since_first, time_since_last, throughput
  FROM stg
  ON CONFLICT (ts, job_name, pid) DO NOTHING
  RETURNING 1
)
SELECT count(*) FROM ins;
COMMIT;
SQL
)
RC=$?

if [ "$RC" -ne 0 ]; then
  echo "$(date '+%F %T') ERREUR psql (code $RC)" >&2
  exit 1
fi
echo "$(date '+%F %T') lues=$NBCSV nouvelles_inserees=${INSERTED:-?} (fenetre: $DAYS)"
