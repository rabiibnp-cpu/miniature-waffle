#!/usr/bin/env bash
# ingestion_stats_to_db.sh  (version jdbc-cli)
# clush (tous les serveurs) -> parsing des logs ingestion -> insertion PostgreSQL via jdbc-cli
# Connexion : fichier newh_aps.conf (lu par jdbc-cli), plus de psql ni de ~/.pgpass.
# Une ligne deja presente (meme ts + job_name + pid) n'est JAMAIS reinseree
# (INSERT ... ON CONFLICT DO NOTHING sur la cle primaire).
#
# Cron (toutes les 5 min) :
#   */5 * * * * /chemin/ingestion_stats_to_db.sh >> /chemin/ingestion_stats_to_db.log 2>&1
#
# Prerequis : clush, gawk, jdbc-cli. Placer newh_aps.conf a cote du script
# (le script fait un cd dans son propre dossier), ou definir CONF / JDBC_CMD.
# Chargement initial de l'historique : LOOKBACK_DAYS=60 ./ingestion_stats_to_db.sh

export PATH="/usr/local/bin:/usr/bin:/bin:$PATH"
cd "$(dirname "$0")" || exit 1

# ---------- Configuration ----------
CONF="${CONF:-newh_aps.conf}"
JDBC_CMD="${JDBC_CMD:-jdbc-cli -c $CONF -ttd -q}"
TABLE="${TABLE:-aps.ingestion_metrics}"
DB_TZ="${DB_TZ:-Europe/Paris}"          # fuseau des dates ecrites dans les logs
LOGDIR="${LOGDIR:-/logs/ingestion}"
CLUSH_OPTS="${CLUSH_OPTS:--a}"
LOOKBACK_DAYS="${LOOKBACK_DAYS:-1}"     # aujourd'hui + N jours precedents (gere le passage de minuit)
BATCH="${BATCH:-400}"                   # lignes par INSERT (limite d'un argument shell ~128 Ko)

# ---------- Anti-chevauchement (si un run dure plus de 5 min) ----------
exec 9>/tmp/ingestion_to_db.lock
flock -n 9 || { echo "$(date '+%F %T') deja en cours, on saute"; exit 0; }

# ---------- Fenetre de dates ----------
DAYS="$(date +%F)"
for i in $(seq 1 "$LOOKBACK_DAYS"); do DAYS="$DAYS|$(date -d "-$i day" +%F)"; done
PATTERN="^($DAYS).*DriverTaskQueueService"
MTIME=$((LOOKBACK_DAYS + 1))            # ne lire que les fichiers modifies recemment

TMPD="$(mktemp -d)"
trap 'rm -rf "$TMPD"' EXIT
CSV="$TMPD/stats.csv"
VALS="$TMPD/values.sql"
ERR="$TMPD/jdbc.err"

# ---------- awk 1 : logs -> CSV ----------
cat > "$TMPD/parse.awk" <<'AWK_EOF'
function secs(s,   t, n, u) {
  t = 0;
  while (match(s, /^[0-9]+[DHMS]/)) {
    n = substr(s, 1, RLENGTH-1) + 0;
    u = substr(s, RLENGTH, 1);
    t += n * (u=="D" ? 86400 : (u=="H" ? 3600 : (u=="M" ? 60 : 1)));
    s = substr(s, RLENGTH+1);
  }
  return t;
}
BEGIN {
  print "date,job,pid,workers_running,workers_failed,items_submitted,items_inflight,results_received,sub_results_received,errors,time_since_submitted,time_since_first_worker,time_since_last_result,throughput";
}
{
  gsub(/\r/, "");
  gsub(/\([+-][0-9]+\)/, "");
  if (!match($0, /([0-9]{4}-[0-9]{2}-[0-9]{2} [0-9:]{8}),[0-9]+ .*DriverTaskQueueService - \[([^#]+)#([^]]+)\] w:([0-9]+)\/([0-9]+) - i:([0-9]+)\/([0-9]+) - r:([0-9]+)\/([0-9]+)\/([0-9]+) - t:([^\/]+)\/([^\/]+)\/([^ ]+)/, m)) next;
  d = m[1]; pid = m[3];
  x = d; gsub(/[-:]/, " ", x);
  ts = mktime(x);
  thr = 0;
  if ((pid in pfl) && ts > pts[pid]) thr = (pfl[pid] - m[7]) / (ts - pts[pid]);
  pfl[pid] = m[7];
  pts[pid] = ts;
  printf "%s,%s,%s,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%.2f\n", d, m[2], pid, m[4], m[5], m[6], m[7], m[8], m[9], m[10], secs(m[11]), secs(m[12]), secs(m[13]), thr;
}
AWK_EOF

# ---------- awk 2 : CSV -> tuples SQL (un par ligne) ----------
# La colonne sub_results_received (CSV col. 9) n'existe pas dans la table : ignoree.
cat > "$TMPD/tosql.awk" <<'AWK_EOF'
BEGIN { FS = ","; q = "\047" }
NR == 1 { next }
{
  job = $2; gsub(q, q q, job);
  pid = $3; gsub(q, q q, pid);
  ts = "(TIMESTAMP " q $1 q " AT TIME ZONE " q tz q ")";
  print "(" ts "," q job q "," q pid q "," $4 "," $5 "," $6 "," $7 "," $8 "," $10 "," $11 "," $12 "," $13 "," $14 ")";
}
AWK_EOF

# ---------- 1) Collecte + parsing -> CSV temporaire ----------
REMOTE="find $LOGDIR -name 'ingestion.log*' -mtime -$MTIME -print0 | xargs -0 -r zgrep -h -E '$PATTERN' 2>/dev/null"

clush $CLUSH_OPTS -N "$REMOTE" \
  | LC_ALL=C sort -s -k1,2 \
  | gawk -f "$TMPD/parse.awk" > "$CSV"

NBCSV=$(( $(wc -l < "$CSV") - 1 ))
if [ "$NBCSV" -le 0 ]; then
  echo "$(date '+%F %T') aucune ligne trouvee (fenetre: $DAYS)"
  exit 0
fi

gawk -v tz="$DB_TZ" -f "$TMPD/tosql.awk" "$CSV" > "$VALS"
split -l "$BATCH" "$VALS" "$TMPD/chunk."

# ---------- 2) Insertion via jdbc-cli, par lots ----------
TOTAL=0
for f in "$TMPD"/chunk.*; do
  ROWS="$(paste -sd, "$f")"
  QUERY="WITH ins AS (INSERT INTO $TABLE (ts, job_name, pid, workers_running, workers_failed, items_submitted, items_inflight, results_received, results_errors, time_since_sub, time_since_first, time_since_last, throughput) VALUES $ROWS ON CONFLICT (ts, job_name, pid) DO NOTHING RETURNING 1) SELECT count(*) FROM ins"

  OUTPUT=$($JDBC_CMD "$QUERY" 2>"$ERR")
  RC=$?
  N=$(printf '%s\n' "$OUTPUT" | grep -Eo '[0-9]+' | tail -n 1)

  if [ "$RC" -ne 0 ] || [ -z "$N" ]; then
    echo "$(date '+%F %T') ERREUR jdbc-cli (code $RC) sortie='$OUTPUT'" >&2
    cat "$ERR" >&2
    exit 1
  fi
  TOTAL=$((TOTAL + N))
done

echo "$(date '+%F %T') lues=$NBCSV nouvelles_inserees=$TOTAL (fenetre: $DAYS)"
